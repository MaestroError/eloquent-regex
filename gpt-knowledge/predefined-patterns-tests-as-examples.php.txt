it('matches valid credit card numbers correctly', function () {
    $string = "Visa: 4111 1111 1111 1111, MasterCard: 5500 0000 0000 0004, Amex: 3400 000000 00009

        Visa: 4111-1111-1111-1111, MasterCard: 5500-0000-0000-0004, Amex: 3400-000000-00009
        
        Visa: 4111111111111111, MasterCard: 5500000000000004, Amex: 340000000000009";
    $builder = EloquentRegex::source($string);
    $matches = $builder->creditCardNumber()->get();

    expect($matches)->toEqual([
        '4111 1111 1111 1111',
        '5500 0000 0000 0004',
        '3400 000000 00009',
        '4111-1111-1111-1111',
        '5500-0000-0000-0004',
        '3400-000000-00009',
        '4111111111111111',
        '5500000000000004',
        '340000000000009',
    ]);
});

it('does not match invalid credit card numbers', function () {
    $string = "Invalid: 1234 5678 9012 3456, Another: 0000 0000 0000 0000, Another 2400 000000 00009";
    $builder = EloquentRegex::source($string);
    $matches = $builder->creditCardNumber()->get();

    expect($matches)->toBeEmpty();
});

it('checks path using pathType option', function () {
    $string = "/usr/local/bin/script.sh";
    $builder = EloquentRegex::source($string);

    // With int
    $check = $builder->filePath(0, null, false, 1)->check(); // 1 for absolute path
    expect($check)->toBeTrue();
    // With string
    $check = $builder->filePath(0, null, false, "absolute")->check();
    expect($check)->toBeTrue();
    // With array
    $check = $builder->filePath(["pathType" => "absolute"])->get();
    expect($check)->toEqual(["/usr/local/bin/script.sh"]);
});

it('validates emails case insensitively', function () {
    $builder = EloquentRegex::source("Example@Email.COM");
    $checkWithoutFlag = $builder->start()
        ->lowercaseText()
        ->character("@")
        ->lowercaseText()
        ->dot()
        ->uppercaseText()
        ->end()
        ->check();

    expect($checkWithoutFlag)->toBeFalse();

    $builder = EloquentRegex::source("Example@Email.COM");
    $checkWithFlag = $builder->start()
        ->lowercaseText()
        ->character("@")
        ->lowercaseText()
        ->dot()
        ->uppercaseText()
        ->end()
        ->asCaseInsensitive()->check();

    expect($checkWithFlag)->toBeTrue();
});
